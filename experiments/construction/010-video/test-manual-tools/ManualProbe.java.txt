import java.nio.ByteBuffer;
import java.nio.file.*;
import java.util.*;
import pt.uminho.esr.ott.streaming.StreamPacket;
import pt.uminho.esr.ott.protocol.MessageCodec;
import pt.uminho.esr.ott.protocol.ProtocolException;

/** Manual artifact only: compiled in a temporary directory, production loaded from the pinned JAR. */
class ManualProbe {
    static int checks;
    static final Scanner CONSOLE = new Scanner(System.in);
    static final UUID STREAM = UUID.fromString("00000000-0000-4000-8000-000000000001");
    static final UUID SESSION = UUID.fromString("deadbeef-0000-4000-8000-000000000001");
    static Path fixtures;
    interface Action { void run() throws Exception; }
    static void check(String label, boolean ok) {
        if (!ok) throw new AssertionError(label);
        checks++;
        System.out.println("CHECK_PASS " + checks + " " + label);
    }
    static boolean rejects(Action action) throws Exception {
        try { action.run(); return false; } catch (ProtocolException expected) { return true; }
    }
    static byte[] fixture(String name) throws Exception { return Files.readAllBytes(fixtures.resolve("udp-" + name + ".bin")); }
    static void stage(int number, String text) {
        System.out.println("STAGE " + number + "/8 " + text + " — confirma e prime Enter");
        if (!CONSOLE.hasNextLine() || !CONSOLE.nextLine().isEmpty())
            throw new IllegalStateException("Manual interrupted: expected Enter");
        System.out.println("OPERATOR_CONFIRM_STAGE=" + number);
    }
    public static void main(String[] args) throws Exception {
        fixtures = Path.of(args[1]);
        stage(1, "Verifica o JAR fixado, origem das classes e versões separadas UDP/TCP");
        Path jar = Path.of(args[0]).toRealPath();
        Path origin = Path.of(StreamPacket.class.getProtectionDomain().getCodeSource().getLocation().toURI()).toRealPath();
        Path tcpOrigin = Path.of(MessageCodec.class.getProtectionDomain().getCodeSource().getLocation().toURI()).toRealPath();
        System.out.println("STREAM_PACKET_ORIGIN=" + origin);
        check("production classes loaded from pinned JAR", jar.equals(origin) && jar.equals(tcpOrigin));
        check("OTT1/UDP v1", StreamPacket.MAGIC == 0x4f545431 && StreamPacket.VERSION == 1);
        check("TCP v2 remains separate", MessageCodec.VERSION == 2 && StreamPacket.HEADER_BYTES == 64);

        stage(2, "Confere o datagrama mínimo: 65 bytes e payload aa");
        byte[] min = HexFormat.of().parseHex("4f54543101000040" + "00000000000040008000000000000001"
                + "deadbeef000040008000000000000001" + "0000000000000000" + "0000000000000000"
                + "00000001" + "00000001" + "aa");
        var value = new StreamPacket(STREAM, SESSION, 0, 0, 1, 0, 1, new byte[]{(byte)0xaa});
        byte[] wire = StreamPacket.encode(value);
        System.out.println("MIN_HEX=" + HexFormat.of().formatHex(wire));
        check("literal minimum equals independent fixture", Arrays.equals(min, fixture("min")));
        check("encoded minimum equals literal 65 bytes", wire.length == 65 && Arrays.equals(wire, min));
        var decoded = StreamPacket.decode(min);
        check("decoded identities/counters/payload", decoded.streamId().equals(STREAM) && decoded.sessionId().equals(SESSION)
                && decoded.frameSeq() == 0 && decoded.mediaTimeUs() == 0 && decoded.frameLength() == 1
                && decoded.fragmentIndex() == 0 && decoded.fragmentCount() == 1 && decoded.payload()[0] == (byte)0xaa);

        stage(3, "Confere fixture small: seq7, tempo10000, payload ABC");
        var small = StreamPacket.decode(fixture("small"));
        check("small metadata", small.frameSeq() == 7 && small.mediaTimeUs() == 10000 && small.frameLength() == 3);
        check("small payload ABC", Arrays.equals(small.payload(), new byte[]{65,66,67}));
        check("independent small encoding", Arrays.equals(fixture("small"), StreamPacket.encode(
                new StreamPacket(STREAM, SESSION, 7, 10000, 3, 0, 1, new byte[]{65,66,67}))));

        stage(4, "Confere fronteira1137: dois fragmentos, payload1136+1");
        byte[] frame = new byte[1137]; Arrays.fill(frame, (byte)0xab); frame[1136] = (byte)0xcd;
        var fragments = StreamPacket.fragment(STREAM, SESSION, 8, 40000, frame);
        check("1137 split golden bytes", fragments.size() == 2 && Arrays.equals(StreamPacket.encode(fragments.get(0)), fixture("split-first"))
                && Arrays.equals(StreamPacket.encode(fragments.get(1)), fixture("split-last")));
        check("fragment index/count", fragments.get(0).fragmentIndex() == 0 && fragments.get(1).fragmentIndex() == 1
                && fragments.get(0).fragmentCount() == 2 && fragments.get(1).fragmentCount() == 2);
        byte[] rebuilt = new byte[1137]; System.arraycopy(fragments.get(0).payload(),0,rebuilt,0,1136);
        System.arraycopy(fragments.get(1).payload(),0,rebuilt,1136,1);
        check("independent1137 recombination", Arrays.equals(frame, rebuilt));

        stage(5, "Confere limite2MiB:1847 fragmentos, último96, datagramas até1200");
        byte[] large = new byte[2097152]; for(int i=0;i<large.length;i++) large[i]=(byte)(i*37+11);
        var pieces = StreamPacket.fragment(STREAM, SESSION, 9, 12345, large);
        check("maximum count/last", pieces.size() == 1847 && pieces.getLast().payload().length == 96);
        boolean capped=true; byte[] joined = new byte[large.length]; int offset=0;
        for(var piece:pieces) { byte[] encoded=StreamPacket.encode(piece); int len=encoded.length-64;
            capped &= encoded.length<=1200 && len>=1 && len<=1136;
            System.arraycopy(encoded,64,joined,offset,len); offset+=len; }
        check("all datagrams physically capped", capped && offset == 2097152);
        check("maximum independent recombination", Arrays.equals(large,joined));

        stage(6, "Confere rejeição dos12 negativos, u32 hostil e overflow de região");
        String[] negatives={"short-header","no-payload","magic","version","flags","header-length","index","count",
                "zero-frame","large-frame","negative-seq","datagram-size"};
        boolean rejected=true; for(String name:negatives) rejected &= rejects(() -> StreamPacket.decode(fixture("invalid-"+name)));
        check("all12 independent negatives rejected",rejected);
        byte[] hostile=min.clone(); ByteBuffer.wrap(hostile).putInt(56,0xffffffff);
        check("u32 maximum rejected",rejects(() -> StreamPacket.decode(hostile)));
        check("region integer overflow rejected",rejects(() -> StreamPacket.decode(min,1,Integer.MAX_VALUE)));

        stage(7, "Confere região exata, buffer maior e byte extra");
        byte[] backing=new byte[1400]; System.arraycopy(fixture("small"),0,backing,17,67);
        check("exact nonzero-offset region",Arrays.equals(StreamPacket.decode(backing,17,67).payload(),new byte[]{65,66,67}));
        check("whole oversized buffer rejected",rejects(() -> StreamPacket.decode(backing)));
        byte[] extra=Arrays.copyOf(min,66);
        check("surplus payload byte rejected",rejects(() -> StreamPacket.decode(extra)));

        stage(8, "Confere cópias defensivas e lista imutável");
        byte[] owned={(byte)0xaa}; var immutable=new StreamPacket(STREAM,SESSION,0,0,1,0,1,owned); owned[0]=0;
        check("input copied",immutable.payload()[0]==(byte)0xaa);
        byte[] exposed=immutable.payload(); exposed[0]=0; byte[] receive=fixture("small"); var received=StreamPacket.decode(receive);receive[64]=0;
        check("output/receive buffer isolated",immutable.payload()[0]==(byte)0xaa && received.payload()[0]==65);
        boolean unmodifiable=false;try{fragments.add(immutable);}catch(UnsupportedOperationException expected){unmodifiable=true;}
        check("immutable fragment list",unmodifiable);
        if(checks!=24) throw new AssertionError("Expected24 checks, got"+checks);
        System.out.println("PROBE_PASS CHECKS="+checks);
    }
}
